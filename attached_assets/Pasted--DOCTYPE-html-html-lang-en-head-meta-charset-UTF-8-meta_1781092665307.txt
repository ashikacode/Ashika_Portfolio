        <p class="card-description">
          Memory in a Scent is a speculative HCI project exploring olfaction as an underutilised sense
          in technology. Using an Arduino Uno, an RGB sensor, and ultrasonic atomisers, the device reads
          the dominant colours in a photograph and releases a corresponding scent — triggering emotional
          memory recall. Designed to be approachable and portable — wood and acrylic, no medical
          equipment, no bulk — it was conceived with dementia and Alzheimer's patients in mind,
          sitting at the intersection of speculative design, health, and human-computer interaction.

<script>
  // Scent particle animation
  const canvas = document.getElementById('scentCanvas');
  const ctx = canvas.getContext('2d');

  function resize() {
    canvas.width  = canvas.offsetWidth;
    canvas.height = canvas.offsetHeight;
  }
  resize();
  window.addEventListener('resize', resize);

  const particles = [];
  const COLORS = ['rgba(126,202,195,', 'rgba(224,69,123,', 'rgba(242,237,228,'];

  for (let i = 0; i < 55; i++) {
    particles.push({
      x:     Math.random() * canvas.width,
      y:     Math.random() * canvas.height,
      r:     Math.random() * 2.5 + 0.5,
      vx:    (Math.random() - 0.5) * 0.3,
      vy:    -(Math.random() * 0.5 + 0.2),
      alpha: Math.random() * 0.35 + 0.05,
      color: COLORS[Math.floor(Math.random() * COLORS.length)],
      life:  Math.random(),
      decay: Math.random() * 0.003 + 0.001,
    });
  }

  function tick() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    particles.forEach(p => {
      p.x += p.vx + Math.sin(p.life * 3) * 0.2;
      p.y += p.vy;
      p.life += p.decay;

      if (p.y < -10 || p.life > 1) {
        p.x    = Math.random() * canvas.width;
        p.y    = canvas.height + 10;
        p.life = 0;
        p.alpha = Math.random() * 0.35 + 0.05;
      }

      const fade = p.life < 0.2 ? p.life / 0.2 : p.life > 0.8 ? (1 - p.life) / 0.2 : 1;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
      ctx.fillStyle = p.color + (p.alpha * fade) + ')';
      ctx.fill();
    });
    requestAnimationFrame(tick);
  }
  tick();
</script>
